What is AutoMapper?

--> AutoMapper is a library that automatically maps one object to another. 
--> It eliminates repetitive code like:

// WITHOUT AutoMapper (Manual mapping - 10+ lines)
var fieldDto = new FieldDto();
fieldDto.Id = field.Id;
fieldDto.FieldName = field.FieldName;
fieldDto.Location = field.Location;
fieldDto.AreaHectares = field.AreaHectares;
fieldDto.SoilType = field.SoilType.ToString();
fieldDto.Status = field.Status.ToString();
fieldDto.CreatedAt = field.CreatedAt;
fieldDto.UpdatedAt = field.UpdatedAt;

// WITH AutoMapper (1 line)
var fieldDto = _mapper.Map<FieldDto>(field);

--> Flow of mapping
┌──────────────────┐     AutoMapper     ┌──────────────────┐
│     Field        │                    │    FieldDto      │
│ (Database Entity)│                    │   (API Response) │
├──────────────────┤                    ├──────────────────┤
│ Id = 1           │ ─────────────────► │ Id = 1           │
│ FieldName = "North"│                   │ FieldName = "North"│
│ SoilType = LOAMY │ ──┐                 │ SoilType = "LOAMY"│
│ Status = ACTIVE  │ ──┼─► ToString()   │ Status = "ACTIVE" │
│ Farm.FarmName    │ ──┘                 │ FarmName = "GF"   │
└──────────────────┘                    └──────────────────┘

--> AutoMapper automatically maps properties with the same name and type. 
--> You only need to write explicit mappings for exceptions (different names, different types, custom logic).

// Field Entity (Source)           // FieldDto (Destination)
public class Field                public class FieldDto
{                                 {
    public int Id { get; set; }       public int Id { get; set; }      // ✅ AUTO mapped
    public int FarmId { get; set; }   public int FarmId { get; set; }  // ✅ AUTO mapped
    public string FieldName { get; set; }  public string FieldName { get; set; } // ✅ AUTO mapped
    public string? Location { get; set; }  public string? Location { get; set; } // ✅ AUTO mapped
    public decimal? AreaHectares { get; set; } // ✅ AUTO mapped
    public SoilTypeEnum? SoilType { get; set; } // ❌ NEEDS mapping (Enum → string)
    public FieldStatusEnum? Status { get; set; } // ❌ NEEDS mapping (Enum → string)
    public DateTime CreatedAt { get; set; }      // ✅ AUTO mapped
    public DateTime? UpdatedAt { get; set; }     // ✅ AUTO mapped
    public virtual Farm? Farm { get; set; }      // ❌ NEEDS mapping (Navigation property)
}                                 }

-> How AutoMapper Works Without Explicit Mapping
// AutoMapper's Default Behavior
CreateMap<Field, FieldDto>()
// It automatically maps ALL properties that:
// 1. Have the SAME name
// 2. Have compatible types

// These properties are auto-mapped without any code:
// Id → Id (int to int)
// FarmId → FarmId (int to int)  
// FieldName → FieldName (string to string)
// Location → Location (string to string)
// AreaHectares → AreaHectares (decimal? to decimal?)
// CreatedAt → CreatedAt (DateTime to DateTime)
// UpdatedAt → UpdatedAt (DateTime? to DateTime?)


--> When You MUST Write Explicit Mapping

1. Enum to String (Different Type)


// Entity has Enum, DTO has String
src.SoilType is SoilTypeEnum (CLAY, SANDY, etc.)
dest.SoilType is string ("CLAY", "SANDY", etc.)

// AutoMapper cannot convert automatically
.ForMember(dest => dest.SoilType, 
    opt => opt.MapFrom(src => src.SoilType.ToString()))

 
--> Different Property Names

csharp
// Entity                                 // DTO
public string FirstName { get; set; }     public string Name { get; set; }

// AutoMapper doesn't know FirstName → Name
.ForMember(dest => dest.Name, 
    opt => opt.MapFrom(src => src.FirstName))


-->Formatting or Transformation

// Entity                                           // DTO
public DateTime CreatedAt { get; set; }        public string CreatedDate { get; set; }

// Need to format date
.ForMember(dest => dest.CreatedDate, 
    opt => opt.MapFrom(src => src.CreatedAt.ToString("yyyy-MM-dd")))



-->Complete Example Showing Auto vs Manual Mapping


public class FieldMappingProfile : Profile
{
    public FieldMappingProfile()
    {
        CreateMap<Field, FieldDto>()
            // ❌ These NEED explicit mapping (exceptions)
            .ForMember(dest => dest.SoilType, 
                opt => opt.MapFrom(src => src.SoilType.ToString()))  // Enum → string
            .ForMember(dest => dest.Status, 
                opt => opt.MapFrom(src => src.Status.ToString()))    // Enum → string
            .ForMember(dest => dest.FarmName, 
                opt => opt.MapFrom(src => src.Farm != null ? src.Farm.FarmName : string.Empty)) // Navigation
            
            // ✅ These are AUTO-MAPPED (no code needed!)
            // Id → Id
            // FarmId → FarmId
            // FieldName → FieldName
            // Location → Location
            // AreaHectares → AreaHectares
            // CreatedAt → CreatedAt
            // UpdatedAt → UpdatedAt
            // (and any other matching properties)
            
            // Note: You don't need to write anything for these!
            ;
    }
}
